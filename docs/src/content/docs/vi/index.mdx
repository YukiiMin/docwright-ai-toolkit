import { Card, CardGrid } from '@astrojs/starlight/components';

## Khả năng cốt lõi

<CardGrid stagger>
  <Card title="Động cơ Word (DOCX)" icon="document">
    **Chuẩn hóa OpenXML ECMA-376**, khóa `<w:cantSplit/>` chống rách hàng qua trang, tự động lặp tiêu đề `<w:tblHeader/>`, và bộ ghép run phẫu thuật chống mất style.
  </Card>
  <Card title="Động cơ Bảng tính (XLSX)" icon="pencil">
    **Đột biến khối 2D với dịch chuyển công thức AST**, nhận diện ốc đảo bảng (Table Islands), bảo vệ logo DrawingML, và kiểm định qua **13 Cổng phổ quát (`UG-01..13`)**.
  </Card>
  <Card title="Sơ đồ kỹ thuật (Draw.io)" icon="seti:graphql">
    Sinh XML thuần `mxGraphModel` với định tuyến trực giao (Orthogonal routing), tự động co giãn kích thước theo trường, và render PNG/SVG siêu nét.
  </Card>
  <Card title="Tích hợp toàn diện mọi AI Agent" icon="laptop">
    Máy chủ STDIO chuẩn **Model Context Protocol (MCP)** + Điều phối CLI. Cấu hình 1-Click cho Cursor, Claude Code, Google Antigravity, Windsurf, Codex CLI.
  </Card>
</CardGrid>
